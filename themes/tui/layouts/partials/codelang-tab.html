{{- $h := .highlighted | string -}} {{- $lang := lower (.lang | default "") -}} {{- if in (slice "console" "bash" "sh" "shell" "zsh") $lang -}} {{- $lines := slice -}} {{- range split $h `` -}} {{- $l := . -}} {{- if not (strings.Contains $l `class="go"`) -}} {{- $l = replaceRE `(function +)([\w.:-]+)` `${1}${2}` $l -}} {{- $l = replaceRE `(\s*)([\w.:-]+)( *\(\))` `${1}${2}${3}` $l -}} {{- $l = replaceRE `((?:\s*(?:[^<]* )?|(?:\||;|&&|\|\|) |(?:do|then|else) |\$\()(?:\w+=(?:[^<]*|[^\s<])* )*)([\w.~/][\w.~/-]*)` `${1}${2}` $l -}} {{- $l = replaceRE `(\s)(--?[A-Za-z][\w-]*)` `${1}${2}` $l -}} {{- end -}} {{- $lines = $lines | append $l -}} {{- end -}} {{- $h = delimit $lines `` -}} {{- else if in (slice "ruby" "rb" "erb") $lang -}} {{- $h = replaceRE `(\d[\d_]*)\.(\d[\d_]*)` `${1}.${2}` $h -}} {{- else if in (slice "javascript" "js" "typescript" "ts" "jsx" "tsx") $lang -}} {{- $h = replaceRE `(function\s+)` `${1}` $h -}} {{- else if in (slice "python" "py" "python3") $lang -}} {{- $h = replaceRE `([^<]+)(\()` `${1}${2}` $h -}} {{- end -}} {{- with .lang -}} {{- $label := index hugo.Data.codelang (lower .) | default . -}} {{- $tab := printf `%s` $label -}} {{- $h = replaceRE `^(
